<script>
// ── Estado ────────────────────────────────────────────────────
var SCANNER     = null;
var PROCESSANDO = false;
var QR_CODES    = new Set();
var BIPADOS     = new Set();
var INFO        = {};  // { nrPedido, sigla, nomeCliente, qtdRolos, idPedido, idCarregamento, nomeUsuario }
var LOG_COUNT   = 0;
var CANAL       = null;

// ── Inicialização ─────────────────────────────────────────────
window.onload = function() {
  // Lê parâmetros da URL: ?nr=...&cli=...&total=...&qrs=...&idP=...&idC=...&op=...
  var params = new URLSearchParams(window.location.search);

  INFO.nrPedido      = params.get('nr')    || '—';
  INFO.nomeCliente   = params.get('cli')   || '—';
  INFO.qtdRolos      = parseInt(params.get('total') || '0');
  INFO.idPedido      = params.get('idP')   || '';
  INFO.idCarregamento= params.get('idC')   || '';
  INFO.nomeUsuario   = params.get('op')    || '';

  // QR Codes esperados chegam como lista separada por vírgula
  var qrsParam = params.get('qrs') || '';
  if (qrsParam) {
    qrsParam.split(',').forEach(function(q) { if (q) QR_CODES.add(q.trim()); });
  }

  document.getElementById('hdr-nr').textContent  = 'Pedido ' + INFO.nrPedido;
  document.getElementById('hdr-cli').textContent  = INFO.nomeCliente;
  document.getElementById('pg-total').textContent = INFO.qtdRolos;

  // Abre canal de comunicação com o Index.html
  try {
    CANAL = new BroadcastChannel('expedicao_scanner');
  } catch(e) {
    console.warn('BroadcastChannel não suportado:', e);
  }

  iniciarScanner();
};

// ── Scanner ───────────────────────────────────────────────────
function iniciarScanner() {
  document.getElementById('scanner-area').style.display  = 'flex';
  document.getElementById('status-bar').style.display    = 'block';
  document.getElementById('tela-erro-cam').style.display = 'none';

  Html5Qrcode.getCameras()
    .then(function(cameras) {
      if (!cameras || cameras.length === 0) {
        mostrarErroCam();
        return;
      }

      // Prefere câmera traseira
      var cameraId = cameras[cameras.length - 1].id;
      for (var i = 0; i < cameras.length; i++) {
        var label = (cameras[i].label || '').toLowerCase();
        if (label.indexOf('back') !== -1 || label.indexOf('rear') !== -1 ||
            label.indexOf('traseira') !== -1 || label.indexOf('environment') !== -1) {
          cameraId = cameras[i].id;
          break;
        }
      }

      SCANNER = new Html5Qrcode('qr-reader');
      SCANNER.start(
        cameraId,
        { fps: 12, qrbox: { width: 220, height: 220 } },
        onQrLido,
        function() {}
      )
      .then(function() {
        document.getElementById('status-bar').textContent = 'Aponte para o QR Code do rolo';
      })
      .catch(function(err) {
        console.error('start error:', err);
        mostrarErroCam();
      });
    })
    .catch(function(err) {
      console.error('getCameras error:', err);
      mostrarErroCam();
    });
}

function tentarNovamente() {
  if (SCANNER) {
    SCANNER.stop().catch(function(){}).finally(function() {
      SCANNER = null;
      iniciarScanner();
    });
  } else {
    iniciarScanner();
  }
}

function mostrarErroCam() {
  if (SCANNER) { SCANNER.stop().catch(function(){}); SCANNER = null; }
  document.getElementById('scanner-area').style.display  = 'none';
  document.getElementById('status-bar').style.display    = 'none';
  document.getElementById('tela-erro-cam').style.display = 'flex';
}

// ── Leitura de QR Code ────────────────────────────────────────
function onQrLido(qrCode) {
  if (PROCESSANDO) return;
  PROCESSANDO = true;
  if (SCANNER) SCANNER.pause();

  processar(qrCode);

  setTimeout(function() {
    try { if (SCANNER) SCANNER.resume(); } catch(e) {}
    PROCESSANDO = false;
  }, 1400);
}

function processar(qrCode) {
  // QR inválido (não pertence a este pedido)
  if (!QR_CODES.has(qrCode)) {
    flash('er');
    showToast('QR inválido — não é deste pedido', 'er');
    addLog('er', 'Inválido: ' + qrCode);
    enviarBip(qrCode, null, 'erro', 'QR Code não pertence a este pedido');
    return;
  }

  // Já bipado
  if (BIPADOS.has(qrCode)) {
    flash('er');
    showToast('Rolo já foi bipado!', 'wa');
    addLog('wa', 'Duplicado: ' + qrCode);
    enviarBip(qrCode, null, 'erro', 'Rolo já bipado');
    return;
  }

  // Bip válido
  BIPADOS.add(qrCode);
  var partes = qrCode.split('-');
  var nrRolo = partes[partes.length - 1];
  var total  = INFO.qtdRolos;

  flash('ok');
  showToast('✓ Rolo ' + nrRolo + '/' + total, 'ok');
  addLog('ok', 'Rolo ' + nrRolo + '/' + total + ' — ' + qrCode);
  atualizarProgresso();
  enviarBip(qrCode, nrRolo, 'ok', 'Rolo ' + nrRolo + '/' + total + ' bipado');

  // Notifica o Index.html via BroadcastChannel
  if (CANAL) {
    CANAL.postMessage({ tipo: 'bip', qrCode: qrCode, nrRolo: nrRolo, resultado: 'ok' });
  }

  // Todos bipados
  if (BIPADOS.size >= INFO.qtdRolos) {
    setTimeout(function() {
      if (SCANNER) { SCANNER.stop().catch(function(){}); SCANNER = null; }
      // Notifica conclusão
      if (CANAL) { CANAL.postMessage({ tipo: 'concluido', total: BIPADOS.size }); }
      mostrarConcluido();
    }, 800);
  }
}

// Salva bip na planilha via google.script.run
function enviarBip(qrCode, nrRolo, resultado, mensagem) {
  google.script.run
    .withFailureHandler(function(e) { console.error('Erro ao salvar bip:', e); })
    .registrarBip({
      idPedido:       INFO.idPedido,
      idCarregamento: INFO.idCarregamento,
      qrCode:         qrCode,
      nrRolo:         nrRolo,
      resultado:      resultado,
      mensagem:       mensagem,
      nomeUsuario:    INFO.nomeUsuario
    });
}

// ── UI ────────────────────────────────────────────────────────
function atualizarProgresso() {
  var pct = INFO.qtdRolos > 0 ? Math.round(BIPADOS.size / INFO.qtdRolos * 100) : 0;
  document.getElementById('pg-feitos').textContent  = BIPADOS.size;
  document.getElementById('pg-fill').style.width    = pct + '%';
}

function mostrarConcluido() {
  document.getElementById('scanner-area').style.display  = 'none';
  document.getElementById('status-bar').style.display    = 'none';
  document.getElementById('tela-concluido').style.display = 'flex';
  document.getElementById('pg-fill').classList.add('ok');
  document.getElementById('concl-sub').textContent =
    BIPADOS.size + ' rolos bipados — Pedido ' + INFO.nrPedido;
}

function fecharScanner() {
  if (SCANNER) { SCANNER.stop().catch(function(){}); SCANNER = null; }
  if (CANAL) { CANAL.postMessage({ tipo: 'fechou' }); CANAL.close(); }
  window.close();
}

function addLog(tipo, texto) {
  var hora = new Date().toLocaleTimeString('pt-BR', { hour:'2-digit', minute:'2-digit', second:'2-digit' });
  var lista = document.getElementById('log-list');
  var item  = document.createElement('div');
  item.className = 'log-item';
  item.innerHTML =
    '<span class="ldot ' + tipo + '"></span>' +
    '<span class="log-txt">' + texto + '</span>' +
    '<span class="log-hr">' + hora + '</span>';
  lista.insertBefore(item, lista.firstChild);
  LOG_COUNT++;
  document.getElementById('log-cnt').textContent = LOG_COUNT;
}

function flash(tipo) {
  var el = document.getElementById('flash');
  el.className = 'flash ' + tipo + ' on';
  setTimeout(function() { el.className = 'flash'; }, 280);
}

function showToast(msg, tipo) {
  var el = document.getElementById('bip-toast');
  el.textContent = msg;
  el.className = 'show ' + tipo;
  clearTimeout(el._t);
  el._t = setTimeout(function() { el.className = ''; }, 2500);
}
